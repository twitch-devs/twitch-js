Getting Started

npm install twitch-js
import TwitchJs from 'twitch-js'

const { chat, api } = new TwitchJs({
  username: 'your_username',
  token: 'your_oauth_token'
})

await chat.connect()
await chat.join('#channel')

chat.on('PRIVMSG', (msg) => {
  console.log(`${msg.username}: ${msg.message}`)
})
